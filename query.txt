-- ================= Horizontal Partition: user nam / nữ =================
CREATE TABLE user_male (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100),
    email VARCHAR(100),
    age INT
);

CREATE TABLE user_female (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100),
    email VARCHAR(100),
    age INT
);

-- ================= Vertical Partition: chi tiết user =================
CREATE TABLE user_detail (
    user_id INT PRIMARY KEY,
    gender VARCHAR(10),
    address VARCHAR(255),
    phone VARCHAR(20)
);

-- ================= Functional Partition: Order / Payment / Shipping =================
CREATE TABLE orders (
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT,
    order_date DATETIME,
    total DECIMAL(10,2)
);

CREATE TABLE payments (
    id INT PRIMARY KEY AUTO_INCREMENT,
    order_id INT,
    payment_date DATETIME,
    amount DECIMAL(10,2),
    method VARCHAR(50)
);

CREATE TABLE shipments (
    id INT PRIMARY KEY AUTO_INCREMENT,
    order_id INT,
    shipping_date DATETIME,
    status VARCHAR(50)
);
